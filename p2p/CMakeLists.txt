cmake_minimum_required(VERSION 3.24)
project(ds4_p2p_standalone LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(DS4_P2P_BUILD_PROBE "Build the real-hardware P2P probe (never runs it)" OFF)
option(DS4_P2P_USE_HIP "Use HIP instead of CUDA for the probe" ON)
set(DS4_P2P_HIP_COMPAT_DIR "" CACHE PATH "Path to Strata's include/strata/hip_compat directory")
set(DS4_P2P_STRATA_INCLUDE_DIR "" CACHE PATH "Path to Strata's include directory")

add_library(ds4_p2p STATIC ds4_p2p.cpp)
target_include_directories(ds4_p2p PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

include(CTest)
if(BUILD_TESTING)
    add_executable(ds4_p2p_unit_test tests/ds4_p2p_test.cpp)
    target_link_libraries(ds4_p2p_unit_test PRIVATE ds4_p2p)
    add_test(NAME ds4_p2p_unit_test COMMAND ds4_p2p_unit_test)
    add_executable(fused_ar_logic_test tests/fused_ar_logic_test.cpp)
    target_include_directories(fused_ar_logic_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME fused_ar_logic_test COMMAND fused_ar_logic_test)
endif()

if(NOT MSVC)
    target_compile_options(ds4_p2p PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING)
        target_compile_options(ds4_p2p_unit_test PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(fused_ar_logic_test PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()

if(DS4_P2P_BUILD_PROBE)
    if(NOT DS4_P2P_STRATA_INCLUDE_DIR AND DS4_P2P_HIP_COMPAT_DIR)
        get_filename_component(DS4_P2P_STRATA_INCLUDE_DIR "${DS4_P2P_HIP_COMPAT_DIR}/../.." ABSOLUTE)
    endif()
    if(NOT EXISTS "${DS4_P2P_STRATA_INCLUDE_DIR}/strata/kernels/f16_bits.hpp")
        message(FATAL_ERROR "Set DS4_P2P_STRATA_INCLUDE_DIR to Strata's include directory, or set DS4_P2P_HIP_COMPAT_DIR to its strata/hip_compat subdirectory")
    endif()
    if(DS4_P2P_USE_HIP)
        if(NOT EXISTS "${DS4_P2P_HIP_COMPAT_DIR}/cuda_runtime.h")
            message(FATAL_ERROR "Set DS4_P2P_HIP_COMPAT_DIR to Strata's include/strata/hip_compat directory")
        endif()
        if(NOT DEFINED CMAKE_HIP_ARCHITECTURES OR CMAKE_HIP_ARCHITECTURES STREQUAL "")
            set(CMAKE_HIP_ARCHITECTURES gfx906 CACHE STRING "HIP target architecture" FORCE)
        endif()
        enable_language(HIP)
        find_package(hip CONFIG REQUIRED)
        set_source_files_properties(ds4_p2p_runtime.cu p2p_probe.cu fused_ar.cu fused_ar_probe.cu p2p_flag_probe.cu
                                    hip_error_probe.cu PROPERTIES LANGUAGE HIP)
        add_library(ds4_p2p_runtime STATIC ds4_p2p_runtime.cu)
        target_include_directories(ds4_p2p_runtime BEFORE PUBLIC ${DS4_P2P_HIP_COMPAT_DIR})
        target_include_directories(ds4_p2p_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(ds4_p2p_runtime PUBLIC STRATA_USE_HIP=1 STRATA_HIP_WAVE64=1)
        target_link_libraries(ds4_p2p_runtime PUBLIC hip::host)
    else()
        enable_language(CUDA)
        find_package(CUDAToolkit REQUIRED)
        add_library(ds4_p2p_runtime STATIC ds4_p2p_runtime.cu)
        target_include_directories(ds4_p2p_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
        target_link_libraries(ds4_p2p_runtime PUBLIC CUDA::cudart)
    endif()
    add_library(ds4_fused_ar STATIC fused_ar.cu)
    target_link_libraries(ds4_fused_ar PUBLIC ds4_p2p ds4_p2p_runtime)
    add_executable(p2p_probe p2p_probe.cu)
    target_link_libraries(p2p_probe PRIVATE ds4_p2p ds4_p2p_runtime)
    set(DS4_FUSED_PROBE_ELEMENTS 16384 CACHE STRING "FP32 elements per member in the fused AR probe")
    add_executable(fused_ar_probe fused_ar_probe.cu)
    target_compile_definitions(fused_ar_probe PRIVATE DS4_FUSED_PROBE_ELEMENTS=${DS4_FUSED_PROBE_ELEMENTS})
    target_include_directories(fused_ar_probe PRIVATE ${DS4_P2P_STRATA_INCLUDE_DIR})
    target_link_libraries(fused_ar_probe PRIVATE ds4_fused_ar ds4_p2p ds4_p2p_runtime)
    add_executable(p2p_flag_probe p2p_flag_probe.cu)
    target_link_libraries(p2p_flag_probe PRIVATE ds4_p2p ds4_p2p_runtime)
    add_executable(hip_error_probe hip_error_probe.cu)
    target_link_libraries(hip_error_probe PRIVATE ds4_p2p_runtime)
endif()
